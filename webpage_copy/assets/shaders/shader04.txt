#extension GL_OES_standard_derivatives : enable

precision highp float;

uniform float time;
uniform vec2 mouse;
uniform vec2 tres;
varying vec2 tcoord;
uniform vec4 seed;
uniform vec4 audio;
uniform vec4 color;
uniform vec4 par_a;
uniform vec4 par_b;

uniform sampler2D tex[8];
uniform int tex_l;


float gwn(vec2 p, vec2 q, vec2 n)
{
	vec2 d = p - q;
	
	return dot(d,n) / (4.0*3.1415925*pow(length(d),3.0));
}


void main( void ) {

	vec2 p = ( gl_FragCoord.xy / tres.xy );// + mouse / 4.0;

	vec4 c = vec4(0);
	
	float wn = 0.0;
	wn += gwn(vec2(0.55, 0.45), p, mouse - 0.5);
	wn += gwn(vec2(0.25, 0.45), p, mouse - 0.5);
	
	float wnsdf = (wn - 0.5) * -1.0;
	
	wnsdf = sign(wnsdf) * pow(abs(wnsdf), 1.0);
	
	float sw = 2.0;
	vec3 tint = (step(vec3(0), vec3(wnsdf) * vec3(-1,0.0,1)) + step(vec3(0), vec3(wnsdf) * vec3(0,1,-1))) *5.;

	c = vec4(vec3(mod(abs(wnsdf) * sw, 1.0)) * tint, 1.0);
	
	c.b = abs(wnsdf) < 0.01 ? 1.0 : 0.0;
	
//	c = vec4(vec3(wn + 0.5),1.0);

	gl_FragColor = c;

}
