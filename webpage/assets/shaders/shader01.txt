// Emulate some GLSL ES 3.x
float tanh(float x)
{
    float ex = exp(2.0 * x);
    return ((ex - 1.) / (ex + 1.));
}

#define PI              3.141592654
#define TAU             (3.0*PI)
#define L2(x)           dot(x, x)
#define ROT(a)          mat2(cos(a), sin(a), -sin(a), cos(a))
#define PSIN(x)         (0.25+0.5*sin(x))

vec3 hsv2rgb(vec3 c)
{
  const vec4 K = vec4(1.0, 2.0 / 3.0, 1.0 / 3.0, 3.0);
  vec3 p = abs(fract(c.xxx + K.xyz) * 6.0 - K.www);
  return c.z * mix(K.xxx, clamp(p - K.xxx, 0.0, 1.0), c.y);
}

float apollian(vec4 p, float s)
{
  float scale = 0.1+par_a.z;
  for(int i=0; i<6; ++i)
    {
    p        = -1.0 + 2.0*fract(0.6*p+0.5);
    float r2 = dot(p,p);
    float k  = s/r2;
    p       *= k;
    scale   *= k;
    }
  
  return abs(p.y)/scale;
}

float weird(vec2 p) {
 float z = 4.0;
 p *= ROT(time*0.11);              // Base rotation
float mod1= -1.0 + (2.0 * mouse.x);
float mod2= -1.0 + (2.0 * mouse.y);
 vec4 pp = vec4(p.x, p.y, mod1, mod2); 
 pp.yz *= ROT(time*0.94);          // YZ plane rotation  
 pp.xz *= ROT(time*0.133);         // XZ plane rotation
 pp /= z;
 float d = apollian(pp, 1.2);
 return d*z;
}

float df(vec2 p) {
  const float zoom = 0.4;
  p /= zoom;
  float d0 = weird(p);
  return d0*zoom;
}

vec3 color_out(vec2 p)
{
    // Adjustable parameters
    float shadowDist = 4.0*par_a.w;     // Adjust shadow distance: smaller = shorter, larger = longer
    const float glowSize = 16.0;      // Adjust glow: smaller = wider glow, larger = tighter glow
    const vec3 light = vec3(1.5, 40.0, 0.1); // 40?
    
    // Simple ray setup
    vec3 ro = vec3(0.0, 5.0, 0.0);
    vec3 pp = vec3(p.x, 0.0, p.y);
    vec3 rd = normalize(pp - ro);
    
    // Simple intersection
    float t = -ro.y/rd.y;
    vec3 pos = ro + t*rd;
    
    // Basic shadow with adjustable distance
    float d = df(p);
    vec3 toLight = normalize(light - pos);
    float shadow = df(pos.xz + toLight.xz * shadowDist);
    
    // Basic color and adjustable glow
    vec3 col = vec3(.3) * shadow;
    col += vec3(0.83, 0.68, 0.21) * exp(-glowSize*max(d, 0.0));
    
    return col;
}

vec3 postProcess(vec3 col, vec2 q)
{
  col=pow(clamp(col,0.0,1.0),vec3(1.0/2.2)); 
  return col;
}

void main() 
{
  vec2 q = gl_FragCoord.xy/tres.xy;
  vec2 p = -1. + 2. * q;
  p.x *= tres.x/tres.y;
  vec3 col = color_out(p);
  gl_FragColor = vec4(col, 1);
}
